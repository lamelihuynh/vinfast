-- round trip 1
SELECT o.id, o.type, o.status, o.payment_status, o.total_amount, o.note, o.created_at,
       COUNT(*) OVER() AS total_rows
FROM orders o
WHERE o.user_id = :uid
ORDER BY o.created_at DESC, o.id DESC
LIMIT :limit OFFSET :offset;

-- round trip 2
SELECT oi.order_id, oi.product_id, oi.variant_id, oi.product_name, oi.variant_name,
       oi.color_name, oi.unit_price, oi.quantity
FROM order_items oi
WHERE oi.order_id = ANY(CAST(:ids AS bigint[]))   -- bind '{12,15,18}'
ORDER BY oi.order_id, oi.id;

-- getById: 1 round trip
SELECT o.*, u.name AS user_name, u.email,
       COALESCE(json_agg(json_build_object(
           'product_id', oi.product_id, 'variant_id', oi.variant_id,
           'product_name', oi.product_name, 'variant_name', oi.variant_name,
           'color_name', oi.color_name, 'unit_price', oi.unit_price, 'quantity', oi.quantity)
         ORDER BY oi.id) FILTER (WHERE oi.id IS NOT NULL), '[]') AS items
FROM orders o
JOIN users u ON u.id = o.user_id
LEFT JOIN order_items oi ON oi.order_id = o.id
WHERE o.id = :id
GROUP BY o.id, u.id;

final class ActivityLog
{
    public const COLLECTION = 'userActivityLogs';
    public const ACTIONS = ['view_product', 'search', 'add_to_cart', 'remove_from_cart',
                            'wishlist_toggle', 'place_order'];

    /** Không bao giờ ném lỗi. $data: productId, variantId, keyword, orderId, resultCount... */
    public static function record(?int $userId, string $actionType, array $data = []): void
    {
        try {
            if (!in_array($actionType, self::ACTIONS, true)) { return; }
            Db::mongo()->selectCollection(self::COLLECTION)->insertOne([
                'userId'     => $userId,
                'sessionId'  => session_id() ?: null,
                'actionType' => $actionType,
                'productId'  => isset($data['productId']) ? (string)$data['productId'] : null,
                'keyword'    => isset($data['keyword']) ? self::normalize($data['keyword']) : null,
                'meta'       => array_diff_key($data, ['productId' => 1, 'keyword' => 1]),
                'timestamp'  => new MongoDB\BSON\UTCDateTime(),
            ], ['writeConcern' => new MongoDB\Driver\WriteConcern(1, 200)]);
        } catch (\Throwable $e) {
            error_log('[ActivityLog] ' . $e->getMessage());
        }
    }

    // topKeywords(), recent(), countByAction(): xem T3-15
}

Quy trình thí nghiệm

Quy trình giống T1: cùng giới hạn container, seed cố định, 5 lần khởi động + 30 lần đo, báo cáo p50/p95. Không đo khi bật Xdebug, DB dùng named volume.

Lệnh

# 1. Dựng dataset
docker compose up -d --build
docker compose exec app composer db:reset -- --size=100k
docker compose exec -T postgres sh -c 'psql -U "$POSTGRES_USER" -d "$POSTGRES_DB" -X' < bench/t3/setup.sql
docker compose exec -T mongo mongosh vinfast --quiet < bench/t3/setup_mongo.js

# 2. Plan PG (lưu JSON)
docker compose exec -T postgres sh -c 'psql -U "$POSTGRES_USER" -d "$POSTGRES_DB" -X -A -t \
  -c "EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) $(cat)"' \
  < bench/t3/queries/top10_join.sql > bench/results/t3_plans/top10_join_100k.json

# 3. Explain Mongo
docker compose exec -T mongo mongosh vinfast --quiet --eval '
  printjson(db.orders_bench.explain("executionStats").aggregate([
    { $match: { status: { $ne: "cancelled" } } }, { $unwind: "$items" },
    { $group: { _id: "$items.productId", qty: { $sum: "$items.quantity" } } },
    { $sort: { qty: -1 } }, { $limit: 10 } ]))' > bench/results/t3_plans/top10_mongo_100k.json

# 4. Chạy đo hàng loạt (ghi CSV)
bash bench/t3/run_sql.sh   --dataset=100k --warmup=5 --runs=30
bash bench/t3/run_mongo.sh --dataset=100k --warmup=5 --runs=30
docker compose exec app php bench/t3/order_history.php --variant=snapshot --user=1234 --runs=30 --dataset=100k

execution_ms lấy từ plan JSON (trong DB), elapsed_ms đo phía client. Ghi lại work_mem, shared_buffers, max_parallel_workers_per_gather và giữ nguyên giữa các biến thể. Biến thể ép join: thêm SET enable_hashjoin = off; trước EXPLAIN.

Report và demo

Dàn ý chương Query processing

  1. Bối cảnh. Vì sao products rời PG làm hỏng các JOIN cũ (Order.php:25-28, DashboardAdminController.php:125, 157, 205), cách snapshot trong order_items giải quyết.
  2. Các node trong plan PG. Seq Scan, Index Scan, Index Only Scan, Bitmap Heap Scan, Sort, HashAggregate vs GroupAggregate, Limit, Gather; đọc một plan JSON thật của top10_join.
  3. Cost ước lượng vs thực tế. Cost là đơn vị tương đối, không phải ms; so est_rows với actual_rows, chỗ nào lệch và vì sao (thống kê, điều kiện <>).
  4. Thuật toán join. Hash Join, Merge Join, Nested Loop: dùng plan ép bằng enable_*, giải thích khi nào planner chọn từng loại.
  5. Subquery vs JOIN vs CTE vs MV. PG biến IN (subquery) thành semi join; subquery tương quan chạy lặp; CTE inline từ PG 12 trừ khi MATERIALIZED; MV đọc nhanh nhưng dữ liệu cũ và tốn refresh.
  6. Pipeline Mongo. Các stage $match (dùng index được), $unwind, $group, $sort, $limit; đọc explain("executionStats"); so với PG trên cùng dữ liệu.
  7. Join ở backend vs join trong DB. Snapshot (2 round trip) vs SQL + $in (3) vs N+1; số liệu ms và round trip.
  8. N+1. Code cũ có dạng gần N+1 ở đâu (buildActivities gọi lại 2 query, 11 lần countRows), đã gộp thế nào.
  9. Activity log. Thiết kế document, 2 index, pipeline thống kê từ khóa, vì sao log không được làm hỏng request.
  10. Bảng 8 loại query-update.
  11. Hạn chế. Eventual consistency giữa hai DB (quyết định 13), số đo trên Docker Desktop.
