CREATE TABLE coupons (
  id BIGSERIAL PRIMARY KEY,
  code VARCHAR(32) NOT NULL UNIQUE,
  discount_type VARCHAR(10) NOT NULL CHECK (discount_type IN ('percent','fixed')),
  discount_value NUMERIC(14,2) NOT NULL CHECK (discount_value > 0),
  max_discount NUMERIC(14,2),
  min_order_total NUMERIC(14,2) NOT NULL DEFAULT 0,

final class CartService
{
  public const MAX_QTY = 10;

  public static function add(int $userId, string $productId, string $variantId,
                 int $qty = 1, ?string $color = null): void
  {
    $qty = max(1, min(self::MAX_QTY, $qty));
    $found = Product::findVariants([[$productId, $variantId]]); // T1
    if (empty($found) || empty($found[0]['is_active'])) {
      throw new InvalidArgumentException('Sản phẩm không còn bán.');
    }
    Db::transaction(function (PDO $pdo) use ($userId, $productId, $variantId, $qty, $color) {
      $cartId = Cart::ensureForUser($pdo, $userId);
      Cart::upsertItem($pdo, $cartId, $productId, $variantId, $qty, $color);
    });
    ActivityLog::record($userId, 'cart_add', compact('productId', 'variantId', 'qty'));
  }

  // update($userId, $itemId, $qty): qty <= 0 thì removeItem; remove($userId, $itemId)
}

T4-08

OrderService::place() với Db::transaction

~8h · Phụ thuộc: T4-04, T5 stub atomic · PG MONGO MỚI

Một ranh giới transaction duy nhất cho mọi loại đơn: BEGIN, INSERT orders, INSERT order_items, InventoryService::decrement, coupon UPDATE, DELETE cart_items, COMMIT. Lỗi ở bất kỳ bước nào thì ROLLBACK.

Các bước

  1. Tra giá/tên từ Mongo trước BEGIN (buildLines) để không giữ khóa PG khi chờ Mongo. Giá luôn lấy từ server.
  2. Sắp item theo (product_id, variant_id) để khóa inventory cùng thứ tự, tránh deadlock (hợp đồng T5).
  3. Retry cả transaction tối đa 3 lần khi ConcurrentUpdateException (thống nhất với Phase 0 việc Db::transaction đổi SQLSTATE 40001/40P01 sang exception này). OutOfStockException, lỗi coupon không retry.
  4. failPoint() sau mỗi bước, tắt khi APP_ENV=production.
  5. Chế độ không transaction chỉ cho bench (APP_ENV=bench, T4_NO_TXN=1). Hợp đồng cấm gọi decrement ngoài transaction nên nhánh này dùng câu atomic riêng.
  6. CLI bench/t4/smoke_place.php: đặt 1 đơn 2 item trên seed nhỏ, in order id.

File

  • app/services/OrderService.php MỚI
  • bench/t4/smoke_place.php MỚI

Kiểm tra

  • docker compose exec app php bench/t4/smoke_place.php in order_id=…; có 1 dòng orders, 2 dòng order_items, tồn kho giảm đúng, giỏ của user đó trống.
  • Đặt số lượng lớn hơn tồn kho: nhận OutOfStockException, không có dòng orders mới.
  • APP_ENV=production FAIL_AT=after_items php bench/t4/smoke_place.php: đơn vẫn tạo thành công (hook bị tắt).
final class OrderService
{
  private const MAX_RETRY = 3;

  /** $items: [['product_id'=>'165','variant_id'=>'165-STD','qty'=>1,'color_code'=>'CE18'], ...]
   * $opts : type, payment_status, note (array), coupon_code, cart_id, amount_due */
  public static function place(int $userId, array $items, array $opts = []): int
  {
    if ($items === []) throw new InvalidArgumentException('Không có sản phẩm để đặt.');
    $type = $opts['type'] ?? 'purchase';
    $lines = self::buildLines($items); // Mongo, ngoài transaction
    usort($lines, fn($a, $b) => [$a['product_id'], $a['variant_id']]
                   <=> [$b['product_id'], $b['variant_id']]);

    if (self::noTxnMode()) return self::runSteps(Db::pg(), $userId, $type, $lines, $opts, false);

    for ($attempt = 1; ; $attempt++) {
      try {
        $orderId = Db::transaction(
          fn(PDO $pdo) => self::runSteps($pdo, $userId, $type, $lines, $opts, true));
        break;
      } catch (ConcurrentUpdateException $e) {
        if ($attempt >= self::MAX_RETRY) throw $e;
        usleep(random_int(5, 30) * 1000); // backoff có jitter
      }
    }
    self::afterCommit($userId, $orderId, $lines); // best-effort, quyết định 13
    return $orderId;
  }

  private static function runSteps(PDO $pdo, int $userId, string $type,
                   array $lines, array $opts, bool $inTxn): int
  {
    $subtotal = array_sum(array_column($lines, 'line_total'));
    [$coupon, $discount] = empty($opts['coupon_code'])
      ? [null, 0.0]
      : CouponService::quote($pdo, $opts['coupon_code'], $userId, $subtotal);
    $due = $opts['amount_due'] ?? ($subtotal - $discount); // đặt cọc: số tiền cọc

    $orderId = Order::insertHeader($pdo, [
      'user_id' => $userId, 'type' => $type,
      'payment_status' => $opts['payment_status']
        ?? ($type === 'deposit' ? 'pending_verify' : 'unpaid'),
      'subtotal' => $subtotal, 'discount_amount' => $discount, 'total_amount' => $due,
      'coupon_id' => $coupon['id'] ?? null, 'note' => $opts['note'] ?? null,
    ]);
    self::failPoint('after_order');

    Order::insertItems($pdo, $orderId, $lines);
    self::failPoint('after_items');

    foreach ($lines as $l) {
      $inTxn
        ? InventoryService::decrement($pdo, $l['product_id'], $l['variant_id'], $l['qty'])
        : self::decrementAutocommit($pdo, $l); // chỉ cho bench
    }
    self::failPoint('after_inventory');

    if ($coupon !== null) {
      CouponService::redeem($pdo, (int)$coupon['id'], $userId, $orderId, $discount);
    }
    if (!empty($opts['cart_id'])) {
      Cart::deleteItems($pdo, (int)$opts['cart_id'], $lines); // chỉ xóa dòng đã đặt
    }
    self::failPoint('after_cart_delete');
    return $orderId;
  }

  private static function failPoint(string $point): void
  {
    if ((getenv('APP_ENV') ?: 'production') === 'production') return;
    if (getenv('FAIL_AT') === $point) {
      throw new RuntimeException("FAIL_AT injected at {$point}");
    }
  }

  private static function noTxnMode(): bool
  {
    return getenv('APP_ENV') === 'bench' && getenv('T4_NO_TXN') === '1';
  }
}

final class CouponService
{
  /** @return array{0: array, 1: float} [coupon, discount] */
  public static function quote(PDO $pdo, string $code, int $userId, float $subtotal): array
  {
    $c = Coupon::findActiveByCode($pdo, strtoupper(trim($code)));
    if ($c === null) throw new CouponException('Mã giảm giá không tồn tại hoặc đã tắt.');
    if ($subtotal < (float)$c['min_order_total']) {
      throw new CouponException('Đơn hàng chưa đạt giá trị tối thiểu của mã.');
    }
    $d = $c['discount_type'] === 'percent'
      ? round($subtotal * (float)$c['discount_value'] / 100)
      : (float)$c['discount_value'];
    if ($c['max_discount'] !== null) $d = min($d, (float)$c['max_discount']);
    return [$c, min($d, $subtotal)];
  }

  public static function redeem(PDO $pdo, int $couponId, int $userId, int $orderId, float $amount): void
  {
    $stmt = $pdo->prepare(
      'UPDATE coupons SET used_count = used_count + 1
        WHERE id = :id AND is_active
        AND used_count < quota
        AND now() >= starts_at AND now() < expires_at
      RETURNING used_count'
    );
    $stmt->execute([':id' => $couponId]);
    if ($stmt->fetchColumn() === false) {
      throw new CouponException('Mã giảm giá đã hết lượt hoặc hết hạn.');
    }
    try {
      $pdo->prepare('INSERT INTO coupon_redemptions (coupon_id, user_id, order_id, amount)
               VALUES (?, ?, ?, ?)')
        ->execute([$couponId, $userId, $orderId, $amount]);
    } catch (PDOException $e) {
      if ($e->getCode() === '23505') throw new CouponException('Bạn đã dùng mã này rồi.');
      throw $e;
    }
  }
}

Quy trình thí nghiệm

Mọi lệnh chạy trong container, cùng giới hạn tài nguyên, seed cố định, không bật Xdebug, DB trên named volume (theo phần Rủi ro của kế hoạch tổng). Fixture bench dùng user id từ 900001 trở lên để không lẫn với seed.

-- bench/t4/verify_fail.sql
SELECT
  (SELECT count(*) FROM orders o
  WHERE o.user_id = :uid
    AND NOT EXISTS (SELECT 1 FROM order_items i WHERE i.order_id = o.id)) AS orders_without_items,
  (SELECT count(*) FROM orders WHERE user_id = :uid) AS orders_total,
  (SELECT count(*) FROM order_items i JOIN orders o ON o.id = i.order_id
  WHERE o.user_id = :uid) AS items_total,
  (SELECT stock FROM inventory
  WHERE product_id = :'pid' AND variant_id = :'vid') AS stock_now,
  (SELECT COALESCE(SUM(i.quantity), 0) FROM order_items i JOIN orders o ON o.id = i.order_id
  WHERE o.user_id = :uid AND i.product_id = :'pid') AS qty_ordered,
  (SELECT count(*) FROM cart_items ci JOIN carts c ON c.id = ci.cart_id
  WHERE c.user_id = :uid) AS cart_items_now,
  (SELECT used_count FROM coupons WHERE code = 'FAIL10') AS coupon_used,
  (SELECT count(*) FROM coupon_redemptions WHERE user_id = :uid) AS redemptions;
-- nhất quán khi: orders_without_items = 0 và stock_ban_đầu - stock_now = qty_ordered
-- và (orders_total = 0 => cart_items_now = số item ban đầu)

2. Coupon quota 5, 50 request đồng thời

docker compose exec -e APP_ENV=bench app php bench/t4/coupon_race.php --strategy=naive --n=50 --runs=10
docker compose exec -e APP_ENV=bench app php bench/t4/coupon_race.php --strategy=conditional --n=50 --runs=10
# tùy chọn: --n=10,50,100,200
# kiểm tra sau mỗi run
SELECT c.used_count, c.quota, (SELECT count(*) FROM coupon_redemptions r WHERE r.coupon_id = c.id) AS redemptions
  FROM coupons c WHERE c.code = 'RACE5';

Trước mỗi run: DELETE FROM coupon_redemptions WHERE coupon_id = (SELECT id FROM coupons WHERE code='RACE5'); UPDATE coupons SET used_count = 0 WHERE code='RACE5';. CHECK used_count <= quota vẫn bật; naive ghi giá trị tuyệt đối nên CHECK không phát hiện được, chỉ bảng redemptions cho thấy vượt quota. Đây là điểm cần nói trong report.

Report và demo

Dàn ý chương Transaction

  1. ACID trong đặt hàng. A: rollback khi lỗi (FAIL_AT). C: CHECK stock >= 0, used_count <= quota, UNIQUE giỏ/redemption. I: READ COMMITTED + UPDATE có điều kiện + sắp item. D: WAL khi commit.
  2. Ranh giới transaction. Sơ đồ place(); tra Mongo trước BEGIN; 1 service cho đặt cọc và mua; retry 3 lần, lỗi kho/coupon không retry.
  3. Bằng chứng rollback. Ma trận từ t4_failpoint.csv, biểu đồ, ảnh psql trước/sau.
  4. Nhất quán coupon. Naive vs UPDATE có điều kiện; đánh giá lại WHERE sau khi chờ khóa; CHECK không bắt được lost update khi ghi giá trị tuyệt đối.
  5. Chi phí transaction theo số item.
  6. Vì sao không distributed transaction PG + Mongo. Không có 2PC chung; PG commit trước, totalSold cập nhật sau (eventual consistency); snapshot giữ đơn cũ đúng; reconcile-sold; hướng mở rộng: outbox.
  7. So sánh Mongo (T4-20) và giới hạn (không cổng thanh toán, không giữ hàng tạm).

Definition of Done

  • Luồng auth, profile, đổi mật khẩu, admin user chạy trên PG; tìm user không phân biệt hoa thường; role có CHECK.
  • Giỏ hàng: 1 giỏ mỗi user, không có dòng trùng, phải đăng nhập, badge đúng.
  • Checkout từ giỏ và đặt cọc đều đi qua OrderService::place(); không còn SQL ghi đơn trong controller.
  • Mọi điểm FAIL_AT để lại 0 dòng dở dang, giỏ hàng còn nguyên, tồn kho không đổi (10/10 lần).
  • Coupon RACE5 kết thúc với đúng 5 lượt dùng; used_count không vượt quota (10/10 lần).
  • Coupon hết hạn bị từ chối bên trong transaction, không tạo đơn.
  • Luồng đặt cọc 3 bước và trang xác nhận không thay đổi với người dùng.
  • Snapshot trong order_items ghi đủ tên, variant, màu, đơn giá, số lượng, thành tiền; payment_status là cột thật.
  • CSV bench/results/t4_*.csv và biểu đồ đã commit; bench/README.md có lệnh chạy lại.
  • Chương report và demo 10 phút đã tập dượt.
