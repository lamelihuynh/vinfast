private static function formatProduct(?array $d): ?array {
    if (!$d) return null;
    $specs = $d['attributes'] ?? []; unset($specs['rangeKm']);
    $variants = array_values(array_filter($d['variants'] ?? [], fn($v) => $v['isActive'] ?? true));
    return [
        'id' => (int)$d['productID'], 'product_id' => (string)$d['productID'],
        'category_id' => (int)$d['categoryID'], 'name' => $d['name'], 'slug' => $d['slug'],
        'description' => $d['description'] ?? '', 'price' => (float)$d['basePrice'],
        'specs' => $specs, 'images' => $d['images'] ?? [], 'is_active' => empty($d['isActive']) ? 0 : 1,
        'created_at' => self::toLocal($d['createdAt'] ?? null),          // 'Y-m-d H:i:s'
        'exterior_colors' => $variants[0]['colors'] ?? [],
        'variants' => $variants, 'avg_rating' => (float)($d['avgRating'] ?? 0),
        'total_reviews' => (int)($d['totalReviews'] ?? 0), 'reviews' => $d['reviews'] ?? [],
        'category_name' => self::categoryName((string)$d['categoryID']),
    ];
}

public static function refreshSummary(string $productId): void {
    try {
        [$avg, $count] = ProductReview::summary($productId);
        $top = array_map([self::class, 'toEmbedded'], ProductReview::latestApproved($productId, 5));
        Db::mongo()->products->updateOne(['productID' => $productId],
            ['$set' => ['avgRating' => round((float)$avg, 2), 'totalReviews' => (int)$count, 'reviews' => $top]]);
    } catch (Throwable $e) {
        error_log("review summary $productId: " . $e->getMessage());   // best-effort
    }
}
